Option Explicit

Private tInicio As Double
Private tUltima As Double
Private nTeclas As Long
Private nSinGuardar As Long
Private procesando As Boolean

Private Sub UserForm_Initialize()
    MostrarEstado "LISTO PARA ESCANEAR", "", "", RGB(90, 90, 90), vbWhite
    ActualizarContador
End Sub

Private Sub UserForm_Activate()
    VentanaAbierta = True
    txtScan.SetFocus
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    VentanaAbierta = False
    DetenerTemporizador
End Sub

Private Sub txtScan_KeyPress(ByVal KeyAscii As MSForms.ReturnInteger)
    Dim k As Integer: k = KeyAscii
    ' Fin de código: unas pistolas envían Enter (CR) y otras salto de línea (LF)
    If k = 13 Or k = 10 Then
        KeyAscii = 0
        If k = 13 Then Procesar "ENTER" Else Procesar "LF"
        Exit Sub
    End If
    If nTeclas = 0 Then tInicio = Timer
    nTeclas = nTeclas + 1
    tUltima = Timer
    ' Red de seguridad: si la pistola no envía nada al final, el temporizador
    ' cierra el escaneo por inactividad
    IniciarTemporizador
End Sub

Private Sub txtScan_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If KeyCode = vbKeyReturn Then
        KeyCode = 0
        Procesar "ENTER"
    ElseIf KeyCode = vbKeyTab Then
        KeyCode = 0
        Procesar "TAB"
    ElseIf KeyCode = 74 And Shift = 2 Then   ' Ctrl+J: salto de línea
        KeyCode = 0
        Procesar "LF"
    End If
End Sub

' La llama el temporizador cuando la pistola dejó de enviar caracteres
Public Sub CerrarPorInactividad()
    If procesando Then Exit Sub
    If Len(txtScan.Text) < 4 Then Exit Sub
    ' Si viene escrito a mano se espera el Enter, para no cortar el serial a medias
    If MsPorCaracter() > 150 Then Exit Sub
    Procesar "SIN TECLA"
End Sub

Private Function MsPorCaracter() As Double
    If nTeclas > 1 Then
        MsPorCaracter = (tUltima - tInicio) * 1000 / (nTeclas - 1)
    Else
        MsPorCaracter = 9999
    End If
End Function

Private Sub Procesar(fin As String)
    If procesando Then Exit Sub
    procesando = True
    DetenerTemporizador

    Dim codigo As String: codigo = txtScan.Text
    txtScan.Text = ""
    ' Una pistola entrega los caracteres muy seguidos; a mano son mucho más lentos
    Dim origen As String
    If MsPorCaracter() < 120 Or fin = "SIN TECLA" Then origen = "Pistola" Else origen = "Manual"
    nTeclas = 0

    Dim serial As String: serial = NormalizarSerial(codigo)
    If serial <> "" Then
        Dim detalle As String
        Select Case ProcesarEscaneo(codigo, origen, detalle)
            Case E_ENCONTRADO
                MostrarEstado "ENCONTRADO", serial, detalle, RGB(0, 140, 60), vbWhite
            Case E_DUPLICADO
                MostrarEstado "YA ESCANEADO (DUPLICADO)", serial, detalle, RGB(230, 120, 0), vbWhite
                Beep
            Case E_OTRA
                MostrarEstado "OTRA SUCURSAL", serial, detalle, RGB(200, 0, 0), vbWhite
                Beep
            Case E_NOEXISTE
                MostrarEstado "NO EXISTE EN AURORA", serial, detalle, RGB(200, 0, 0), vbWhite
                Beep
            Case E_SALIDA
                MostrarEstado "EN SALIDA (NV / PICKING)", serial, detalle, RGB(0, 90, 170), vbWhite
                Beep
            Case E_EXCLUIDO
                MostrarEstado "BODEGA QUE NO SE CUENTA", serial, detalle, RGB(100, 100, 100), vbWhite
                Beep
            Case E_SKU
                MostrarEstado "CÓDIGO DE PRODUCTO, NO SERIAL", serial, detalle, RGB(255, 200, 0), vbBlack
                Beep
        End Select
        ActualizarContador

        nSinGuardar = nSinGuardar + 1
        If nSinGuardar >= 25 Then
            GuardarSilencioso
            nSinGuardar = 0
        End If
    End If
    procesando = False
End Sub

Private Sub MostrarEstado(titulo As String, serial As String, detalle As String, fondo As Long, letra As Long)
    lblEstado.Caption = titulo
    lblEstado.BackColor = fondo
    lblEstado.ForeColor = letra
    lblSerial.Caption = serial
    lblDetalle.Caption = detalle
End Sub

Private Sub ActualizarContador()
    Dim esp As Long: esp = dEsperado.Count
    Dim pct As Double
    If esp > 0 Then pct = nEncontrados / esp
    lblContador.Caption = "Encontrados " & nEncontrados & " de " & esp & "  (" & Format(pct, "0%") & ")" & _
        "        Faltan " & (esp - nEncontrados) & vbLf & _
        "Sobrantes " & nSobrantes & "        Duplicados " & nDuplicados & "        En salida " & nSalida
End Sub

Private Sub btnDeshacer_Click()
    DetenerTemporizador
    Dim s As String: s = DeshacerUltimo()
    If s = "" Then
        MostrarEstado "NADA QUE DESHACER", "", "No hay escaneos en la toma actual.", RGB(90, 90, 90), vbWhite
    Else
        MostrarEstado "ESCANEO ANULADO", s, "Se anuló el último escaneo. Si corresponde, vuelva a escanearlo.", RGB(90, 90, 90), vbWhite
        nSinGuardar = nSinGuardar + 1
    End If
    ActualizarContador
    txtScan.Text = ""
    nTeclas = 0
    txtScan.SetFocus
End Sub

Private Sub btnCerrar_Click()
    DetenerTemporizador
    Unload Me
End Sub
